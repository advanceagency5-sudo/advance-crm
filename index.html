الأتمتة وتكاملات WhatsApp CRM

بناء سيناريوهات الرد والمتابعة التلقائية لتقليل الهدر وسرعة الاستجابة، مطابقةً لخدمة "الأتمتة وCRM" التي تقدمها AdVance.

رسالة ترحيبية فورية عبر WhatsApp

عند إضافة عميل محتمل جديد من الموقع أو الإعلانات، يتم إرسال رسالة ترحيب مخصصة مع رابط بروفايل AdVance 2026 ورابط حجز مكالمة اكتشاف أولية.

"أهلاً بك أستاذ {الاسم} في AdVance. سعداء باهتمامك بخدمة {الخدمة}. يسعدنا ترتيب جلسة اكتشاف سريعة لتحديد خطة النمو الأنسب لشركتكم..."

تنبيه المتابعة بعد 24 ساعة

إذا ظل العميل في مرحلة "عميل جديد" لأكثر من 24 ساعة دون تسجيل نشاط أو مكالمة اكتشاف، يتم إرسال إشعار فوري لمسؤول المبيعات لتجنب ضياع الفرصة.

"⚠️ تنبيه: العميل {الشركة} بانتظار التواصل منذ 24 ساعة. يرجى الاتصال أو إرسال تفاصيل العرض."

أتمتة إرسال العرض ومتابعة المراجعة

عند تحويل الصفقة إلى مرحلة "إعداد وتقديم العرض"، يتم إنشاء مهمة متابعة تلقائية بعد 3 أيام عمل للتفاوض ومناقشة تفاصيل العرض الفني والمالي.

"تم إنشاء تذكير تلقائي: متابعة استلام العرض وملاحظات العميل {الشركة} بعد 72 ساعة."

تجديد الشراكات الشهرية (Retainers)

إرسال تقرير الإنجاز الشهري ومؤشرات الأداء (KPIs) للعميل في اليوم 25 من كل شهر لتأكيد استمرار الشراكة ومناقشة توسيع نطاق العمل للمرحلة التالية.

"تقرير الأداء الشهري جاهز لمراجعته مع العميل {الشركة} وتجديد باقة الشراكة للشهر القادم."

function submitAdvanceLead(e) {
  e.preventDefault();
  const data = {
    company_name: document.getElementById('leadCompany').value,
    contact_person: document.getElementById('leadName').value,
    contact_phone: document.getElementById('leadPhone').value,
    primary_service: document.getElementById('leadService').value,
    source: 'website'
  };
  fetch('https://ad-vance.org/api.php', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(data)
  }).then(res => res.json()).then(res => {
    alert('شكراً لك! تم استلام طلبك وسيتواصل معك فريق AdVance خلال 24 ساعة.');
    document.getElementById('advanceLeadForm').reset();
  }).catch(() => {
    // توجيه تلقائي لواتساب كبديل فوري
    window.open('https://wa.me/201148484548?text=' + encodeURIComponent('مرحباً، أود الاستفسار عن خدمات AdVance لشركة ' + data.company_name));
  });
}
</script>
